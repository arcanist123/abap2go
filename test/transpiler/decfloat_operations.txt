*** decfloat16: DIV, the quotient rounds down for a positive divisor and up for a negative one
DATA lv_a TYPE decfloat16.
DATA lv_b TYPE decfloat16.
DATA lv_r TYPE decfloat16.
lv_a = '7.5'.
lv_b = 2.
lv_r = lv_a DIV lv_b.
ASSERT lv_r = 3.
lv_a = '-7.5'.
lv_r = lv_a DIV lv_b.
ASSERT lv_r = -4.
lv_b = -2.
lv_r = lv_a DIV lv_b.
ASSERT lv_r = 4.
lv_a = '7.5'.
lv_r = lv_a DIV lv_b.
ASSERT lv_r = -3.

*** decfloat16: comparisons with decfloat, i, p, f, c and string, the scale does not count
* waits: BETWEEN in conditions
DATA lv_a TYPE decfloat16.
DATA lv_b TYPE decfloat16.
DATA lv_i TYPE i VALUE 2.
DATA lv_p TYPE p LENGTH 8 DECIMALS 2 VALUE '1.25'.
DATA lv_f TYPE f.
DATA lv_c TYPE c LENGTH 5 VALUE '1.250'.
DATA lv_s TYPE string VALUE `-3`.
lv_a = '1.25'.
lv_b = '1.3'.
lv_f = '1.2'.
ASSERT lv_a < lv_b.
ASSERT lv_b > lv_a.
ASSERT lv_a <= '1.25'.
ASSERT lv_a >= 1.
ASSERT lv_a BETWEEN 1 AND lv_b.
ASSERT lv_a < lv_i.
ASSERT lv_a = lv_p.
ASSERT lv_a > lv_f.
ASSERT lv_a = lv_c.
ASSERT lv_a > lv_s.
ASSERT NOT lv_a > lv_b.
lv_b = '1.2500'.
ASSERT lv_a = lv_b.
lv_a = '-0.01'.
ASSERT lv_a < 0.

*** decfloat16: division by zero is CX_SY_ZERODIVIDE and keeps the target, 0 / 0 is 0
* waits: TRY ... CATCH
DATA lv_a TYPE decfloat16.
DATA lv_b TYPE decfloat16.
DATA lv_r TYPE decfloat16.
DATA lv_caught TYPE abap_bool.
lv_a = '1.5'.
lv_r = 5.
TRY.
    lv_r = lv_a / lv_b.
  CATCH cx_sy_zerodivide.
    lv_caught = abap_true.
ENDTRY.
ASSERT lv_caught = abap_true.
ASSERT lv_r = 5.
CLEAR lv_a.
lv_r = lv_a / lv_b.
ASSERT lv_r = 0.

*** decfloat16: SORT and READ TABLE WITH KEY on a decfloat component
* waits: structures and internal tables
TYPES: BEGIN OF ty_row,
         key    TYPE c LENGTH 1,
         amount TYPE decfloat16,
       END OF ty_row.
DATA lt_rows TYPE STANDARD TABLE OF ty_row WITH DEFAULT KEY.
DATA ls_row TYPE ty_row.
DATA lv_keys TYPE string.
ls_row-key = 'B'.
ls_row-amount = '-0.75'.
APPEND ls_row TO lt_rows.
ls_row-key = 'A'.
ls_row-amount = '2.25'.
APPEND ls_row TO lt_rows.
ls_row-key = 'C'.
ls_row-amount = -3.
APPEND ls_row TO lt_rows.
SORT lt_rows BY amount.
LOOP AT lt_rows INTO ls_row.
  lv_keys = lv_keys && ls_row-key.
ENDLOOP.
ASSERT lv_keys = `CBA`.
CLEAR lv_keys.
SORT lt_rows BY amount DESCENDING.
LOOP AT lt_rows INTO ls_row.
  lv_keys = lv_keys && ls_row-key.
ENDLOOP.
ASSERT lv_keys = `ABC`.
READ TABLE lt_rows INTO ls_row WITH KEY amount = '-0.75'.
ASSERT sy-subrc = 0.
ASSERT ls_row-key = 'B'.
READ TABLE lt_rows TRANSPORTING NO FIELDS WITH KEY amount = '-0.7'.
ASSERT sy-subrc = 4.
