*** integer MOD
DATA foo TYPE i.
foo = 5 MOD 2.
ASSERT foo = 1.

*** integer MOD, lower case
DATA foo TYPE i.
foo = 5 mod 2.
ASSERT foo = 1.

*** mod, negative value
DATA n TYPE i.
n = -4 MOD 3.
ASSERT n = 2.

*** MOD, both negative
DATA int TYPE i.
int = -5 MOD -2.
ASSERT int = 1.

*** MOD, negative
DATA int TYPE i.
int = 5 MOD -2.
ASSERT int = 1.

*** MOD, zero and zero
DATA bar TYPE i.
bar = 0 MOD 0.
ASSERT bar = 0.
