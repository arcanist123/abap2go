*** division by zero is CX_SY_ZERODIVIDE and keeps the target, 0 / 0 is 0, text that is no number, also 1E3
* waits: TRY ... CATCH

DATA lv_a TYPE int8 VALUE 7.
DATA lv_b TYPE int8.
DATA lv_r TYPE int8 VALUE 5.
DATA lv_count TYPE i.
TRY.
    lv_r = lv_a / lv_b.
  CATCH cx_sy_zerodivide.
    lv_count = lv_count + 1.
ENDTRY.
ASSERT lv_r = 5.
TRY.
    lv_r = lv_a DIV lv_b.
  CATCH cx_sy_zerodivide.
    lv_count = lv_count + 1.
ENDTRY.
TRY.
    lv_r = lv_a MOD lv_b.
  CATCH cx_sy_zerodivide.
    lv_count = lv_count + 1.
ENDTRY.
ASSERT lv_count = 3.
CLEAR lv_a.
lv_r = lv_a / lv_b.
ASSERT lv_r = 0.
TRY.
    lv_text = 'abc'.
    lv_r = lv_text.
  CATCH cx_sy_conversion_no_number.
    lv_count = lv_count + 1.
ENDTRY.
TRY.
    lv_text = '1E3'.
    lv_r = lv_text.
  CATCH cx_sy_conversion_no_number.
    lv_count = lv_count + 1.
ENDTRY.
ASSERT lv_count = 5.

*** COLLECT sums, SORT and READ TABLE WITH KEY on an int8 component above 2^53
* waits: structures and internal tables
TYPES: BEGIN OF ty_row,
         key    TYPE c LENGTH 1,
         amount TYPE int8,
       END OF ty_row.
DATA lt_rows TYPE STANDARD TABLE OF ty_row WITH DEFAULT KEY.
DATA ls_row TYPE ty_row.
DATA lv_big TYPE int8.
DATA lv_keys TYPE string.
lv_big = 9007199254740993.
ls_row-key = 'B'.
ls_row-amount = lv_big.
COLLECT ls_row INTO lt_rows.
ls_row-key = 'A'.
ls_row-amount = 22.
COLLECT ls_row INTO lt_rows.
ls_row-key = 'B'.
ls_row-amount = 2.
COLLECT ls_row INTO lt_rows.
ls_row-key = 'C'.
ls_row-amount = -30.
COLLECT ls_row INTO lt_rows.
ASSERT lines( lt_rows ) = 3.
READ TABLE lt_rows INTO ls_row WITH KEY key = 'B'.
ASSERT |{ ls_row-amount }| = `9007199254740995`.
SORT lt_rows BY amount.
LOOP AT lt_rows INTO ls_row.
  lv_keys = lv_keys && ls_row-key.
ENDLOOP.
ASSERT lv_keys = `CAB`.
CLEAR lv_keys.
SORT lt_rows BY amount DESCENDING.
LOOP AT lt_rows INTO ls_row.
  lv_keys = lv_keys && ls_row-key.
ENDLOOP.
ASSERT lv_keys = `BAC`.
lv_big = lv_big + 2.
READ TABLE lt_rows INTO ls_row WITH KEY amount = lv_big.
ASSERT sy-subrc = 0.
ASSERT ls_row-key = 'B'.
