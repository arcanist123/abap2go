*** i * i above 2^53
DATA lv_a TYPE i.
DATA lv_b TYPE i.
DATA lv_8 TYPE int8.
lv_a = 94906267.
lv_b = 94906267.
lv_8 = lv_a * lv_b.
ASSERT lv_8 = 9007199515875289.

*** int8 / int8 into int8 rounds half away from zero
DATA lv_x TYPE int8.
DATA lv_y TYPE int8.
DATA lv_8 TYPE int8.
lv_x = 7.
lv_y = 2.
lv_8 = lv_x / lv_y.
ASSERT lv_8 = 4.

lv_8 = lv_x / lv_y.
ASSERT lv_8 = -4.
